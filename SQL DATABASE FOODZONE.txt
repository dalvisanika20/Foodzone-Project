CREATE DATABASE FOODZONE;
USE FOODZONE;

CREATE TABLE customers (
    customer_id INT PRIMARY KEY AUTO_INCREMENT,
    first_name VARCHAR(50) NOT NULL,
    last_name VARCHAR(50) NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    phone VARCHAR(15) UNIQUE NOT NULL,
    address VARCHAR(255),
    city VARCHAR(50),
    registration_date DATE,
    status VARCHAR(20) DEFAULT 'Active'
);

CREATE TABLE restaurants (
    restaurant_id INT PRIMARY KEY AUTO_INCREMENT,
    restaurant_name VARCHAR(100) NOT NULL,
    owner_name VARCHAR(100) NOT NULL,
    phone VARCHAR(15) UNIQUE NOT NULL,
    email VARCHAR(100) UNIQUE,
    address VARCHAR(255),
    city VARCHAR(50),
    food_type VARCHAR(50),
    opening_time TIME,
    closing_time TIME,
    rating DECIMAL(2,1) DEFAULT 0.0,
    status VARCHAR(20) DEFAULT 'Active'
);

CREATE TABLE menu (
    food_id INT PRIMARY KEY AUTO_INCREMENT,
    restaurant_id INT NOT NULL,
    food_name VARCHAR(100) NOT NULL,
    food_type VARCHAR(50),
    price DECIMAL(10,2) NOT NULL,
    available VARCHAR(20) DEFAULT 'Yes',

    FOREIGN KEY (restaurant_id)
    REFERENCES restaurants(restaurant_id)
);

CREATE TABLE orders (
    order_id INT PRIMARY KEY AUTO_INCREMENT,
    customer_id INT NOT NULL,
    restaurant_id INT NOT NULL,
    order_date DATETIME NOT NULL,
    total_amount DECIMAL(10,2),
    order_status VARCHAR(20) DEFAULT 'Placed',

    FOREIGN KEY (customer_id)
        REFERENCES customers(customer_id),

    FOREIGN KEY (restaurant_id)
        REFERENCES restaurants(restaurant_id)
);


CREATE TABLE order_items (
    item_id INT PRIMARY KEY AUTO_INCREMENT,
    order_id INT NOT NULL,
    food_id INT NOT NULL,
    quantity INT NOT NULL,
    price DECIMAL(10,2) NOT NULL,

    FOREIGN KEY (order_id)
    REFERENCES orders(order_id),

    FOREIGN KEY (food_id)
    REFERENCES menu(food_id)
);


CREATE TABLE payments (
    payment_id INT PRIMARY KEY AUTO_INCREMENT,
    order_id INT NOT NULL,
    payment_method VARCHAR(30),
    payment_amount DECIMAL(10,2),
    payment_status VARCHAR(20),
    payment_date DATETIME,

    FOREIGN KEY (order_id)
    REFERENCES orders(order_id)
);


CREATE TABLE deliveries (
    delivery_id INT PRIMARY KEY AUTO_INCREMENT,
    order_id INT NOT NULL,
    partner_name VARCHAR(100),
    partner_phone VARCHAR(15),
    delivery_status VARCHAR(20),
    delivery_time INT,

    FOREIGN KEY (order_id)
    REFERENCES orders(order_id)
);

CREATE TABLE coupons (
    coupon_id INT PRIMARY KEY AUTO_INCREMENT,
    coupon_code VARCHAR(30) UNIQUE,
    discount_percent DECIMAL(5,2),
    start_date DATE,
    end_date DATE,
    status VARCHAR(20)
);

CREATE TABLE reviews (
    review_id INT PRIMARY KEY AUTO_INCREMENT,
    customer_id INT NOT NULL,
    restaurant_id INT NOT NULL,
    order_id INT NOT NULL,
    rating DECIMAL(2,1),
    comment VARCHAR(255),
    review_date DATE,

    FOREIGN KEY (customer_id)
    REFERENCES customers(customer_id),

    FOREIGN KEY (restaurant_id)
    REFERENCES restaurants(restaurant_id),

    FOREIGN KEY (order_id)
    REFERENCES orders(order_id)
);

USE FOODZONE;

CREATE TABLE food_categories (
    category_id INT PRIMARY KEY AUTO_INCREMENT,
    category_name VARCHAR(50) UNIQUE NOT NULL,
    description VARCHAR(255)
);

CREATE TABLE delivery_partners (
    delivery_partner_id INT PRIMARY KEY AUTO_INCREMENT,
    first_name VARCHAR(50) NOT NULL,
    last_name VARCHAR(50) NOT NULL,
    phone VARCHAR(15) UNIQUE NOT NULL,
    email VARCHAR(100) UNIQUE,
    joining_date DATE,
    vehicle_type VARCHAR(30),
    vehicle_number VARCHAR(30),
    city VARCHAR(50),
    status VARCHAR(20) DEFAULT 'Available'
);

INSERT INTO orders
(customer_id, restaurant_id, order_date, total_amount, order_status)
WITH RECURSIVE order_numbers AS (
    SELECT 1 AS n

    UNION ALL

    SELECT n + 1
    FROM order_numbers
    WHERE n < 300
)
SELECT
    ((n - 1) % 50) + 1 AS customer_id,
    ((n - 1) % 15) + 1 AS restaurant_id,

    DATE_ADD(
        '2025-01-01 10:00:00',
        INTERVAL (n * 2) DAY
    ) AS order_date,

    CASE ((n - 1) % 10)
        WHEN 0 THEN 299.00
        WHEN 1 THEN 450.00
        WHEN 2 THEN 599.00
        WHEN 3 THEN 749.00
        WHEN 4 THEN 899.00
        WHEN 5 THEN 349.00
        WHEN 6 THEN 499.00
        WHEN 7 THEN 699.00
        WHEN 8 THEN 999.00
        ELSE 399.00
    END AS total_amount,

    CASE ((n - 1) % 5)
        WHEN 0 THEN 'Delivered'
        WHEN 1 THEN 'Delivered'
        WHEN 2 THEN 'Placed'
        WHEN 3 THEN 'Cancelled'
        ELSE 'Delivered'
    END AS order_status

FROM order_numbers;


INSERT INTO order_items
(order_id, food_id, quantity, price)
WITH RECURSIVE item_numbers AS (
    SELECT 1 AS n

    UNION ALL

    SELECT n + 1
    FROM item_numbers
    WHERE n < 600
)
SELECT
    ((n - 1) % 300) + 1 AS order_id,
    ((n * 7 - 1) % 100) + 1 AS food_id,
    ((n - 1) % 3) + 1 AS quantity,
    (
        SELECT price
        FROM menu
        WHERE food_id = ((n * 7 - 1) % 100) + 1
    ) AS price
FROM item_numbers;


INSERT INTO payments
(order_id, payment_method, payment_amount, payment_status, payment_date)
SELECT
    order_id,

    CASE (order_id % 5)
        WHEN 1 THEN 'UPI'
        WHEN 2 THEN 'Credit Card'
        WHEN 3 THEN 'Debit Card'
        WHEN 4 THEN 'Cash'
        ELSE 'Wallet'
    END AS payment_method,

    total_amount AS payment_amount,

    CASE order_status
        WHEN 'Cancelled' THEN 'Refunded'
        ELSE 'Paid'
    END AS payment_status,

    order_date AS payment_date

FROM orders;


INSERT INTO deliveries
(order_id, partner_name, partner_phone, delivery_status, delivery_time)
SELECT
    o.order_id,

    CONCAT(dp.first_name, ' ', dp.last_name) AS partner_name,

    dp.phone AS partner_phone,

    CASE
        WHEN o.order_status = 'Cancelled' THEN 'Cancelled'
        WHEN o.order_status = 'Delivered' THEN 'Delivered'
        WHEN o.order_id % 3 = 0 THEN 'Out for Delivery'
        ELSE 'Assigned'
    END AS delivery_status,

    CASE
        WHEN o.order_status = 'Cancelled' THEN 0
        ELSE 20 + (o.order_id % 41)
    END AS delivery_time

FROM orders o
JOIN delivery_partners dp
    ON dp.delivery_partner_id = ((o.order_id - 1) % 25) + 1
WHERE o.order_id <= 250;


INSERT INTO reviews
(customer_id, restaurant_id, order_id, rating, comment, review_date)
SELECT
    o.customer_id,
    o.restaurant_id,
    o.order_id,

    CASE (o.order_id % 5)
        WHEN 0 THEN 5.0
        WHEN 1 THEN 4.5
        WHEN 2 THEN 4.0
        WHEN 3 THEN 3.5
        ELSE 4.5
    END AS rating,

    CASE (o.order_id % 5)
        WHEN 0 THEN 'Excellent food and great service'
        WHEN 1 THEN 'Very tasty food and quick delivery'
        WHEN 2 THEN 'Good food and reasonable price'
        WHEN 3 THEN 'Food was good but delivery was late'
        ELSE 'Nice experience, would order again'
    END AS comment,

    DATE(o.order_date) AS review_date

FROM orders o
WHERE o.order_id <= 150;

CREATE FUNCTION CalculateDeliveryCharge(distance DECIMAL(10,2))
RETURNS DECIMAL(10,2)
DETERMINISTIC
BEGIN
    DECLARE charge DECIMAL(10,2);
    IF distance <= 5 THEN
        SET charge = 30;
    ELSEIF distance <= 10 THEN
        SET charge = 50;
    ELSE
        SET charge = 80;
    END IF;
    RETURN charge;
END //

    DECLARE category VARCHAR(20);

75)Trigger 1 – Update Order Total
DELIMITER //
CREATE TRIGGER update_order_total
AFTER INSERT ON order_items
FOR EACH ROW
BEGIN
    UPDATE orders
    SET total_amount = (
        SELECT SUM(quantity * price)
        FROM order_items
        WHERE order_id = NEW.order_id
    )
    WHERE order_id = NEW.order_id;
END //
DELIMITER ;

76)Trigger 2 – Update Order Status After Payment
DELIMITER //
CREATE TRIGGER update_order_status_after_payment
AFTER INSERT ON payments
FOR EACH ROW
BEGIN
    IF NEW.payment_status = 'Paid' THEN
        UPDATE orders
        SET order_status = 'Confirmed'
        WHERE order_id = NEW.order_id;
    END IF;
END //
DELIMITER ;
